function doPost(e) { var corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; try { var workerName = e.parameter.workerName; var auditTime = e.parameter.auditTime; var totalScore = e.parameter.totalScore; var percentage = e.parameter.percentage; var grade = e.parameter.grade; var scores = JSON.parse(e.parameter.scores); var remarks = JSON.parse(e.parameter.remarks || "{}"); // ดักจับข้อมูลรูปภาพ (รองรับทั้งแบบรายข้อ และแบบรูปเดี่ยวส่งท้ายฟอร์ม) var images = {}; try { images = JSON.parse(e.parameter.images || "{}"); } catch(pErr) { images = {}; } var imageBase64 = e.parameter.imageBase64; var imageContentType = e.parameter.imageContentType; var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var scoreArray = []; var allEvidence = []; var firstUploadedImageUrl = "-"; // ตัวแปรสำหรับเก็บลิงก์รูปแรกที่จะเอาไปวาดโชว์ในเซลล์ // ตั้งชื่อหรือค้นหาโฟลเดอร์สำหรับเก็บรูปภาพ var folder; try { var folders = DriveApp.getFoldersByName("QC_Audit_Images"); if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder("QC_Audit_Images"); } folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(fErr) { // ป้องกันกรณีระบบไดรฟ์มีปัญหา } // 1. วนลูปจัดการข้อมูลข้อ 1-14 และอัปโหลดรูปภาพรายข้อ (ระบบเดิมของคุณ) for (var i = 1; i <= 14; i++) { var statusText = (scores[i] === 1) ? "ผ่าน" : (scores[i] === 0 ? "ไม่ผ่าน" : "-"); scoreArray.push(statusText); var remarkText = remarks[i] ? remarks[i].trim() : ""; var imageUrl = "-"; // อัปโหลดรูปภาพข้อ 1-14 (ถ้าหน้าเว็บมีการส่งมา) if (images[i] && images[i].contentType && images[i].base64) { try { var decoded = Utilities.base64Decode(images[i].base64.split(",")[1] || images[i].base64); var blob = Utilities.newBlob(decoded, images[i].contentType, "ข้อ_" + i + "_" + workerName + "_" + Date.now() + ".jpg"); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // แปลงเป็นลิงก์แบบ Direct View เพื่อให้สูตร IMAGE ดึงไปแสดงผลได้ imageUrl = "https://docs.google.com/uc?export=view&id=" + file.getId(); if (firstUploadedImageUrl === "-") { firstUploadedImageUrl = imageUrl; // บันทึกรูปแรกที่เจอไว้ไปแสดงในช่องตาราง } } catch(imgErr) { imageUrl = "Error: " + imgErr.toString(); } } if (remarkText !== "" || imageUrl !== "-") { var evidenceStr = "[ข้อ " + i + "]"; if (remarkText !== "") evidenceStr += " หมายเหตุ: " + remarkText; if (imageUrl !== "-") evidenceStr += " 📸 ลิงก์รูป: " + imageUrl; allEvidence.push(evidenceStr); } } // 2. ตรวจสอบรูปภาพแบบส่งเดี่ยวท้ายฟอร์ม (ถ้าหน้าเว็บใช้โครงสร้างนี้ส่งมาแทน) if (imageBase64 && imageContentType && folder) { try { var decodedSingle = Utilities.base64Decode(imageBase64.split(",")[1] || imageBase64); var blobSingle = Utilities.newBlob(decodedSingle, imageContentType, "QC_Main_" + workerName + "_" + Date.now() + ".jpg"); var fileSingle = folder.createFile(blobSingle); fileSingle.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var singleImageUrl = "https://docs.google.com/uc?export=view&id=" + fileSingle.getId(); allEvidence.push("[รูปภาพหลักฐานหน้างาน] 📸 ลิงก์รูป: " + singleImageUrl); if (firstUploadedImageUrl === "-") { firstUploadedImageUrl = singleImageUrl; } } catch(singleImgErr) { allEvidence.push("[รูปภาพหลักฐาน] Error: " + singleImgErr.toString()); } } // สรุปหลักฐานข้อความทั้งหมดรวมกันเพื่อบันทึกสำรองไว้ในคอลัมน์ V var finalEvidenceText = allEvidence.length > 0 ? allEvidence.join("\n") : "-"; // 3. เรียงลำดับข้อมูลลงแถวตารางใน Google Sheets var rowData = [ new Date(), // คอลัมน์ A: Timestamp auditTime, // คอลัมน์ B: วันที่และเวลาตรวจสอบ workerName // คอลัมน์ C: ชื่อพนักงานกรอ ]; rowData = rowData.concat(scoreArray); // คอลัมน์ D ถึง Q: คำว่า "ผ่าน" หรือ "ไม่ผ่าน" rowData.push(totalScore, percentage, grade); // คอลัมน์ R, S, T: คะแนนรวม, เปอร์เซ็นต์, ผลตรวจ rowData.push(""); // คอลัมน์ U: ล็อกพื้นที่ไว้สำหรับ "ฝังรูปภาพวาดจริง" rowData.push(finalEvidenceText); // คอลัมน์ V: เก็บข้อความหมายเหตุและลิงก์สำรองกันเหนียว sheet.appendRow(rowData); // ---- ✨ ระบบจัดโครงสร้างตารางขยายช่อง และวาดรูปอัตโนมัติ ---- var lastRow = sheet.getLastRow(); // ขยายความสูงของแถวเป็น 90 พิกเซล เพื่อให้รูปภาพแสดงผลได้ขนาดใหญ่ชัดเจน sheet.setRowHeight(lastRow, 90); var dataRange = sheet.getRange(lastRow, 1, 1, 22); // คอลัมน์ A ถึง V dataRange.setFontFamily("Sarabun") .setFontSize(10) .setVerticalAlignment("middle") .setHorizontalAlignment("center") .setWrap(true); // 📌 บังคับสั่งรันสูตร IMAGE แสดงรูปภาพตัวเป็นๆ ทันทีในเซลล์คอลัมน์ U (ถ้ามีรูปภาพส่งมา) if (firstUploadedImageUrl !== "-") { var cellU = sheet.getRange(lastRow, 21); cellU.setFormula('=IMAGE("' + firstUploadedImageUrl + '")'); } else { sheet.getRange(lastRow, 21).setValue("-"); } // จัดตำแหน่งข้อความหมายเหตุสรุปในคอลัมน์ V ให้ชิดซ้ายเพื่อให้อ่านง่าย sheet.getRange(lastRow, 22).setHorizontalAlignment("left"); // ระบบไฮไลต์สีพาสเทลอัตโนมัติ (R-T) var gradeRange = sheet.getRange(lastRow, 18, 1, 3); if (parseInt(totalScore) >= 11) { gradeRange.setBackground("#DCFCE7").setFontColor("#15803D").setFontWeight("bold"); } else { gradeRange.setBackground("#FEE2E2").setFontColor("#B91C1C").setFontWeight("bold"); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON) .setHeaders(corsHeaders); } catch(error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON) .setHeaders(corsHeaders); } } function doOptions(e) { return ContentService.createTextOutput("").setHeaders({ "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }); }
