import pandas as pd # ========================================== # 1. ตั้งค่าชื่อไฟล์และชื่อคอลัมน์ (ปรับเปลี่ยนตรงนี้ได้) # ========================================== input_file = "bank_statement_10_years.xlsx" # หรือ .csv output_file = "financial_summary_report.xlsx" # เปลี่ยนชื่อให้ตรงกับหัวคอลัมน์ในไฟล์ของคุณ col_date = "Date" col_deposit = "Deposit" col_withdraw = "Withdrawal" # ========================================== # 2. โหลดข้อมูลและเตรียมความพร้อม # ========================================== print("กำลังโหลดข้อมูล... อาจใช้เวลาสักครู่เนื่องจากไฟล์มีขนาดใหญ่") # ถ้าเป็นไฟล์ CSV ให้เปลี่ยนเป็น pd.read_csv(input_file) df = pd.read_excel(input_file) # แปลงวันที่ให้อยู่ในรูปแบบที่โปรแกรมเข้าใจ และเรียงลำดับจากอดีตไปปัจจุบัน df[col_date] = pd.to_datetime(df[col_date], errors="coerce") df = df.dropna(subset=[col_date]).sort_values(by=col_date) # เคลียร์ค่าว่าง (NaN) ในคอลัมน์เงินให้เป็น 0 df[col_deposit] = pd.to_numeric(df[col_deposit], errors="coerce").fillna(0) df[col_withdraw] = pd.to_numeric(df[col_withdraw], errors="coerce").fillna(0) # สร้างคอลัมน์ปี และ ปี-เดือน เพื่อใช้จัดกลุ่ม df["Year"] = df[col_date].dt.year df["Year_Month"] = df[col_date].dt.to_period("M") # ========================================== # 3. คำนวณหา "วันที่ฝาก/ถอนมากที่สุด" ของแต่ละเดือน # ========================================== # หารายวันก่อน daily_summary = ( df.groupby([df[col_date].dt.to_period("M"), col_date])[ [col_deposit, col_withdraw] ] .sum() .reset_index() ) # หาวันที่ฝากมากสุดในแต่ละเดือน idx_max_dep = daily_summary.groupby("Date")"Deposit".idxmax() max_dep_day = daily_summary.loc[idx_max_dep, ["Date", col_date, col_deposit]] max_dep_day.columns = ["Year_Month", "Max_Deposit_Date", "Max_Deposit_Amount"] # หาวันที่ถอนมากสุดในแต่ละเดือน idx_max_wth = daily_summary.groupby("Date")"Withdrawal".idxmax() max_wth_day = daily_summary.loc[idx_max_wth, ["Date", col_date, col_withdraw]] max_wth_day.columns = ["Year_Month", "Max_Withdraw_Date", "Max_Withdraw_Amount"] # ========================================== # 4. ประมวลผลภาพรวมรายเดือน (Monthly Summary) # ========================================== print("กำลังประมวลผลสรุปรายเดือน...") # นับจำนวนครั้งที่มีการฝาก หรือ ถอนจริง (ยอดเงิน > 0) df["is_deposit"] = df[col_deposit] > 0 df["is_withdraw"] = df[col_withdraw] > 0 monthly_stats = df.groupby("Year_Month").agg( ยอดฝากรวม=(col_deposit, "sum"), ยอดถอนรวม=(col_withdraw, "sum"), จำนวนครั้งที่ฝาก=("is_deposit", "sum"), จำนวนครั้งที่ถอน=("is_withdraw", "sum"), ) # คำนวณยอดคงเหลือสุทธิของแต่ละเดือน และยอดสะสมยกไป (Cumulative Balance) monthly_stats["ยอดคงเหลือสุทธิประจำเดือน"] = ( monthly_stats["ยอดฝากรวม"] - monthly_stats["ยอดถอนรวม"] ) monthly_stats["ยอดคงเหลือสะสม (End of Month)"] = monthly_stats[ "ยอดคงเหลือสุทธิประจำเดือน" ].cumsum() # นำข้อมูลวันที่ฝาก/ถอนมากที่สุดมารวมเข้าด้วยกัน monthly_report = monthly_stats.reset_index() monthly_report = pd.merge( monthly_report, max_dep_day, on="Year_Month", how="left" ) monthly_report = pd.merge( monthly_report, max_wth_day, on="Year_Month", how="left" ) # จัดรูปแบบการแสดงผลวันที่ให้ดูง่าย (YYYY-MM-DD) monthly_report["Max_Deposit_Date"] = monthly_report[ "Max_Deposit_Date" ].dt.strftime("%Y-%m-%d") monthly_report["Max_Withdraw_Date"] = monthly_report[ "Max_Withdraw_Date" ].dt.strftime("%Y-%m-%d") monthly_report["Year_Month"] = monthly_report["Year_Month"].astype(str) # ========================================== # 5. ประมวลผลภาพรวมรายปี (Yearly Summary) # ========================================== print("กำลังประมวลผลสรุปรายปี...") yearly_report = df.groupby("Year").agg( ยอดฝากรวมประจำปี=(col_deposit, "sum"), ยอดถอนรวมประจำปี=(col_withdraw, "sum"), จำนวนครั้งที่ฝากทั้งปี=("is_deposit", "sum"), จำนวนครั้งที่ถอนทั้งปี=("is_withdraw", "sum"), ) yearly_report["ยอดคงเหลือสุทธิประจำปี"] = ( yearly_report["ยอดฝากรวมประจำปี"] - yearly_report["ยอดถอนรวมประจำปี"] ) yearly_report["ยอดคงเหลือสะสมสุทธิ"] = yearly_report[ "ยอดคงเหลือสุทธิประจำปี" ].cumsum() yearly_report = yearly_report.reset_index() # ========================================== # 6. ส่งออกเป็นไฟล์ Excel แยก Sheets # ========================================== print(f"กำลังบันทึกผลลัพธ์ลงไฟล์: {output_file}") with pd.ExcelWriter(output_file, engine="openpyxl") as writer: monthly_report.to_excel(writer, sheet_name="สรุปรายเดือน", index=False) yearly_report.to_excel(writer, sheet_name="สรุปรายปี", index=False) print("✨ คำนวณเสร็จเรียบร้อยแล้ว! โปรดเปิดดูไฟล์ Excel ที่สร้างขึ้น")
